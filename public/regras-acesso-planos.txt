rules_version = '2';

service cloud.firestore {
  match /databases/{database}/documents {
    function pertence(empresaId) {
      return request.auth != null
        && exists(/databases/$(database)/documents/membros/$(request.auth.uid))
        && get(/databases/$(database)/documents/membros/$(request.auth.uid)).data.ativo == true
        && get(/databases/$(database)/documents/membros/$(request.auth.uid)).data.empresaId == empresaId;
    }

    // EMPLOYEE_PERMISSIONS_V2 — server-managed membership, per-area read/write.
    function responsavel(empresaId) {
      let path = /databases/$(database)/documents/empresas/$(empresaId);
      return exists(path) && 'ownerUid' in get(path).data
        ? get(path).data.ownerUid == request.auth.uid
        : get(/databases/$(database)/documents/membros/$(request.auth.uid)).data.get('papel', '') == 'admin';
    }
    function areaPermite(empresaId, area, escrita) {
      let m = get(/databases/$(database)/documents/membros/$(request.auth.uid)).data;
      let key = m.get('permissionsVersion', 0) == 1 && area in ['orders', 'production'] ? 'budgets' : area;
      let p = m.get('permissoes', {}).get(key, {});
      return pertence(empresaId) && (responsavel(empresaId)
        || (m.get('permissionsVersion', 0) in [1, 2] && p.get('read', false) == true
          && (!escrita || p.get('write', false) == true)));
    }
    function areaColecao(colecao) {
      return colecao == 'sm2_clientes' ? 'clients'
        : colecao == 'sm2_madeiras' ? 'catalog'
        : colecao in ['sm2_estoque1', 'sm2_estoque2', 'sm2_movimentos'] ? 'stock'
        : colecao == 'sm2_carregamentos' ? 'loads'
        : colecao == 'sm2_orcamento_itens' ? 'budgets'
        : colecao == 'sm2_orcamentos_guardados' ? 'orders'
        : colecao == 'sm2_servicos' ? 'services'
        : colecao == 'sm2_resumos_calculadora' ? 'calculator' : 'none';
    }
    function janelaValida(p) {
      return p.endsAt is int && p.graceEndsAt is int && p.consultEndsAt is int
        && p.endsAt > 0 && p.graceEndsAt == p.endsAt + 432000000
        && p.consultEndsAt == p.endsAt + 3024000000;
    }
    function planoPermite(empresaId, escrita) {
      let path = /databases/$(database)/documents/empresas/$(empresaId)/acesso/plano;
      return !exists(path) || get(path).data.enabled != true
        || (janelaValida(get(path).data)
          && request.time.toMillis() < (escrita ? get(path).data.graceEndsAt : get(path).data.consultEndsAt));
    }
    match /empresas/{empresaId}/acesso/plano {
      allow get: if pertence(empresaId);
      allow list, write: if false;
    }

    match /membros/{uid} {
      allow get: if request.auth != null && request.auth.uid == uid;
      allow list, create, update, delete: if false;
    }

    match /empresas/{empresaId}/dados/{colecao}/registros/{id} {
      allow read: if pertence(empresaId) && planoPermite(empresaId, false)
        && ((colecao == 'sm2_sync_status')
          || (colecao in ['sm2_clientes','sm2_madeiras','sm2_estoque1','sm2_estoque2','sm2_orcamentos_guardados','sm2_movimentos','sm2_servicos','sm2_resumos_calculadora']
            && (areaPermite(empresaId, areaColecao(colecao), false)
              || (colecao == 'sm2_madeiras' && (areaPermite(empresaId, 'calculator', false)
                || areaPermite(empresaId, 'budgets', false) || areaPermite(empresaId, 'orders', false)
                || areaPermite(empresaId, 'stock', false) || areaPermite(empresaId, 'loads', false))))));
      allow write: if pertence(empresaId) && planoPermite(empresaId, true)
        && ((colecao == 'sm2_sync_status' && responsavel(empresaId))
          || (colecao in ['sm2_clientes','sm2_madeiras','sm2_estoque1','sm2_estoque2','sm2_movimentos','sm2_servicos','sm2_resumos_calculadora']
            && areaPermite(empresaId, areaColecao(colecao), true)));
    }

    // EMPLOYEE_LOAD_DRAFTS_V1: employees only read and keep drafts in montagem.
    function camposAdministrativosCarga() {
      return ['dadosClienteCargaV407', 'dadosTransporteCargaV414', 'dadosFiscalCargaV559',
        'financeiroOriginalV403', 'financeiroCopiaV403', 'copiaComercialPercentualV400',
        'copiaComercialPercentualV399', 'copiaComercialAtualizadaEmV400',
        'copiaComercialAtualizadaEmV399', 'cargaConferidaPor', 'cargaConferidaPorUid',
        'cargaFechadaPor', 'cargaFechadaPorUid'];
    }
    function nomeFuncionarioCarga() {
      let m = get(/databases/$(database)/documents/membros/$(request.auth.uid)).data;
      return m.get('nome', '') != '' ? m.nome
        : m.get('email', '') != '' ? m.email : request.auth.uid;
    }
    function montagemCarga(d) {
      return d.get('cargaStatus', '') == 'montagem'
        && d.get('cargaFechadaEm', null) == null && d.get('cargaBaixadaEm', null) == null
        && d.get('cargaConferidaPor', '') in ['', null];
    }
    match /empresas/{empresaId}/dados/sm2_carregamentos/registros/{id} {
      allow read: if pertence(empresaId) && planoPermite(empresaId, false)
        && (responsavel(empresaId) || (areaPermite(empresaId, 'loads', false)
          && resource.data.cargaStatus == 'montagem'));
      allow create: if pertence(empresaId) && planoPermite(empresaId, true)
        && (responsavel(empresaId) || (areaPermite(empresaId, 'loads', true)
          && montagemCarga(request.resource.data) && request.resource.data.cargaFluxoV388 == true
          && request.resource.data.grupoCargaId is string && request.resource.data.grupoCargaId.size() > 0
          && request.resource.data.cargaCriadaPorUid == request.auth.uid
          && request.resource.data.cargaCriadaPorNome == nomeFuncionarioCarga()
          && !request.resource.data.keys().hasAny(camposAdministrativosCarga())));
      allow update: if pertence(empresaId) && planoPermite(empresaId, true)
        && (responsavel(empresaId) || (areaPermite(empresaId, 'loads', true)
          && montagemCarga(resource.data) && montagemCarga(request.resource.data)
          && request.resource.data.diff(resource.data).affectedKeys().hasOnly(['atualizadoEm', 'ocultoResumo'])));
      allow delete: if pertence(empresaId) && planoPermite(empresaId, true) && responsavel(empresaId);
    }

    function camposProducao() {
      return ['producaoStatus', 'producaoIniciadaEm', 'producaoPausadaEm', 'producaoRetomadaEm',
        'producaoFinalizadaEm', 'producaoResponsavel', 'observacaoProducao',
        'observacaoProducaoAtualizadaEm', 'materiaisUsadosProducao'];
    }
    match /empresas/{empresaId}/dados/sm2_orcamentos_guardados/registros/{id} {
      allow create: if areaPermite(empresaId, 'orders', true) && planoPermite(empresaId, true)
        && (areaPermite(empresaId, 'production', true)
          || !request.resource.data.keys().hasAny(camposProducao()));
      allow update: if areaPermite(empresaId, 'orders', true) && planoPermite(empresaId, true)
        && (areaPermite(empresaId, 'production', true)
          || !request.resource.data.diff(resource.data).affectedKeys().hasAny(camposProducao()));
      allow delete: if areaPermite(empresaId, 'orders', true) && planoPermite(empresaId, true);
    }

    match /empresas/{empresaId}/dados/sm2_orcamento_itens/registros/{id} {
      allow read: if areaPermite(empresaId, 'budgets', false) && planoPermite(empresaId, false);
      allow delete: if areaPermite(empresaId, 'budgets', true) && planoPermite(empresaId, true);
      allow create, update: if areaPermite(empresaId, 'budgets', true) && planoPermite(empresaId, true)
        && request.resource.data.keys().hasAll(['excluido'])
        && request.resource.data.excluido is bool;
    }

    match /empresas/{empresaId}/dados/sm2_empresas_perfis_dev/registros/{id} {
      allow get: if pertence(empresaId) && id == empresaId;
      allow create, update: if pertence(empresaId) && responsavel(empresaId) && planoPermite(empresaId, true)
        && id == empresaId && request.resource.data.id == empresaId;
      allow list, delete: if false;
    }

    match /{document=**} {
      allow read, write: if false;
    }
  }
}
