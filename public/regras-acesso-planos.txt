rules_version = '2';

service cloud.firestore {
  match /databases/{database}/documents {
    function pertence(empresaId) {
      return request.auth != null
        && exists(/databases/$(database)/documents/membros/$(request.auth.uid))
        && get(/databases/$(database)/documents/membros/$(request.auth.uid)).data.ativo == true
        && get(/databases/$(database)/documents/membros/$(request.auth.uid)).data.empresaId == empresaId;
    }

    // EMPLOYEE_PERMISSIONS_V1 — server-managed membership, per-area read/write.
    function responsavel(empresaId) {
      let path = /databases/$(database)/documents/empresas/$(empresaId);
      return exists(path) && 'ownerUid' in get(path).data
        ? get(path).data.ownerUid == request.auth.uid
        : get(/databases/$(database)/documents/membros/$(request.auth.uid)).data.get('papel', '') == 'admin';
    }
    function areaPermite(empresaId, area, escrita) {
      let m = get(/databases/$(database)/documents/membros/$(request.auth.uid)).data;
      let p = m.get('permissoes', {}).get(area, {});
      return pertence(empresaId) && (responsavel(empresaId)
        || (m.get('permissionsVersion', 0) == 1 && p.get('read', false) == true
          && (!escrita || p.get('write', false) == true)));
    }
    function areaColecao(colecao) {
      return colecao == 'sm2_clientes' ? 'clients'
        : colecao == 'sm2_madeiras' ? 'catalog'
        : colecao in ['sm2_estoque1', 'sm2_estoque2', 'sm2_movimentos'] ? 'stock'
        : colecao == 'sm2_carregamentos' ? 'loads'
        : colecao in ['sm2_orcamento_itens', 'sm2_orcamentos_guardados'] ? 'budgets'
        : colecao == 'sm2_servicos' ? 'services'
        : colecao == 'sm2_resumos_calculadora' ? 'calculator' : 'none';
    }
    function janelaValida(p) {
      return p.endsAt is int && p.graceEndsAt is int && p.consultEndsAt is int
        && p.endsAt > 0 && p.graceEndsAt == p.endsAt + 432000000
        && p.consultEndsAt == p.endsAt + 3024000000;
    }
    function planoPermite(empresaId, escrita) {
      let path = /databases/$(database)/documents/empresas/$(empresaId)/acesso/plano;
      return !exists(path) || get(path).data.enabled != true
        || (janelaValida(get(path).data)
          && request.time.toMillis() < (escrita ? get(path).data.graceEndsAt : get(path).data.consultEndsAt));
    }
    match /empresas/{empresaId}/acesso/plano {
      allow get: if pertence(empresaId);
      allow list, write: if false;
    }

    match /membros/{uid} {
      allow get: if request.auth != null && request.auth.uid == uid;
      allow list, create, update, delete: if false;
    }

    match /empresas/{empresaId}/dados/{colecao}/registros/{id} {
      allow read: if pertence(empresaId) && planoPermite(empresaId, false)
        && ((colecao == 'sm2_sync_status')
          || (colecao in ['sm2_clientes','sm2_madeiras','sm2_estoque1','sm2_estoque2','sm2_carregamentos','sm2_orcamentos_guardados','sm2_movimentos','sm2_servicos','sm2_resumos_calculadora']
            && areaPermite(empresaId, areaColecao(colecao), false)));
      allow write: if pertence(empresaId) && planoPermite(empresaId, true)
        && ((colecao == 'sm2_sync_status' && responsavel(empresaId))
          || (colecao in ['sm2_clientes','sm2_madeiras','sm2_estoque1','sm2_estoque2','sm2_carregamentos','sm2_orcamentos_guardados','sm2_movimentos','sm2_servicos','sm2_resumos_calculadora']
            && areaPermite(empresaId, areaColecao(colecao), true)));
    }

    match /empresas/{empresaId}/dados/sm2_orcamento_itens/registros/{id} {
      allow read: if areaPermite(empresaId, 'budgets', false) && planoPermite(empresaId, false);
      allow delete: if areaPermite(empresaId, 'budgets', true) && planoPermite(empresaId, true);
      allow create, update: if areaPermite(empresaId, 'budgets', true) && planoPermite(empresaId, true)
        && request.resource.data.keys().hasAll(['excluido'])
        && request.resource.data.excluido is bool;
    }

    match /empresas/{empresaId}/dados/sm2_empresas_perfis_dev/registros/{id} {
      allow get: if pertence(empresaId) && id == empresaId;
      allow create, update: if pertence(empresaId) && responsavel(empresaId) && planoPermite(empresaId, true)
        && id == empresaId && request.resource.data.id == empresaId;
      allow list, delete: if false;
    }

    match /{document=**} {
      allow read, write: if false;
    }
  }
}
